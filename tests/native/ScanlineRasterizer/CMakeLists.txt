cmake_minimum_required(VERSION 3.20)
project(JaliumScanlineTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JALIUM_CORE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.core/include")
add_executable(scanline_tests scanline_tests.cpp)
target_include_directories(scanline_tests PRIVATE "${JALIUM_CORE_INCLUDE}")
target_compile_definitions(scanline_tests PRIVATE USE_STROKE_BUFFER=1)
enable_testing()
add_test(NAME scanline_tests COMMAND scanline_tests)
add_executable(path_engine_selection_tests path_engine_selection_tests.cpp)
target_include_directories(path_engine_selection_tests PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.d3d12/include")
add_test(NAME path_engine_selection_tests COMMAND path_engine_selection_tests)
add_executable(frame_latency_selection_tests frame_latency_selection_tests.cpp)
target_include_directories(frame_latency_selection_tests PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.d3d12/include")
add_test(NAME frame_latency_selection_tests COMMAND frame_latency_selection_tests)
add_executable(simple_stroke_tests simple_stroke_tests.cpp)
target_include_directories(simple_stroke_tests PRIVATE "${JALIUM_CORE_INCLUDE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.d3d12/include")
add_test(NAME simple_stroke_tests COMMAND simple_stroke_tests)
add_executable(coverage_mask_tests coverage_mask_tests.cpp)
target_include_directories(coverage_mask_tests PRIVATE "${JALIUM_CORE_INCLUDE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.d3d12/include")
add_test(NAME coverage_mask_tests COMMAND coverage_mask_tests)
add_executable(stroke_mask_cache_tests stroke_mask_cache_tests.cpp)
target_include_directories(stroke_mask_cache_tests PRIVATE "${JALIUM_CORE_INCLUDE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.d3d12/include")
add_test(NAME stroke_mask_cache_tests COMMAND stroke_mask_cache_tests)

add_executable(polyline_flatten_tests polyline_flatten_tests.cpp)
target_include_directories(polyline_flatten_tests PRIVATE "${JALIUM_CORE_INCLUDE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/native/jalium.native.d3d12/include")
add_test(NAME polyline_flatten_tests COMMAND polyline_flatten_tests)
if(REFERENCE_INCLUDE_DIR)
    add_executable(scanline_reference scanline_tests.cpp)
    target_include_directories(scanline_reference PRIVATE "${REFERENCE_INCLUDE_DIR}" "${JALIUM_CORE_INCLUDE}")
    target_compile_definitions(scanline_reference PRIVATE USE_STROKE_BUFFER=0)
endif()
if(MSVC)
    foreach(target scanline_tests scanline_reference simple_stroke_tests coverage_mask_tests polyline_flatten_tests stroke_mask_cache_tests)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE /utf-8)
        endif()
    endforeach()
endif()
